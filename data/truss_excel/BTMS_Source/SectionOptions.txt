Private Sub cboWindLoad_Click()

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".cboWindLoad_Click()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    cboTrussStyle.Enabled = True
    Dim i As Integer, V As Integer
    cboTrussStyle.Clear
    
    If varPassDoor.strPN <> "" Then
        cboTrussStyle.AddItem "No truss"
        cboTrussStyle.ItemData(cboTrussStyle.NewIndex) = 0
        cboTrussStyle.AddItem "Special truss schedule"
        cboTrussStyle.ItemData(cboTrussStyle.NewIndex) = 1000
        cboTrussStyle.ListIndex = 0
        Exit Sub
    End If
    
    If cboWindLoad.ItemData(cboWindLoad.ListIndex) < 3 Then
        If NewLinearHW = False Then
            With cboTrussStyle
            .Clear
            'i = 1
            Call GetStandardTrussSchedule 'get the minimum schedule.
            If DrWidth <= 300 Or ProdType = "Rolltite" Then
                If (varDoorData.TrussSchedule = 0) Then
                    .AddItem "No truss"
                    .ItemData(.NewIndex) = 0
                End If
                'i = (varDoorData.TrussSchedule)
                If (varDoorData.TrussSchedule <= 1) Then
                    .AddItem "1 every top and bottom section"
                    .ItemData(.NewIndex) = 1
                End If
                If (varDoorData.TrussSchedule <= 2) Then
                    .AddItem "1 Top, 1 every other section and 1 bottom"
                    .ItemData(.NewIndex) = 2
                End If
                If (varDoorData.TrussSchedule <= 3) Then
                    .AddItem "1 every section"
                    .ItemData(.NewIndex) = 3
                End If
                If (varDoorData.TrussSchedule <= 4) Then
                    .AddItem "1 every section and 2 every 2nd section"
                    .ItemData(.NewIndex) = 4
                End If
                If (varDoorData.TrussSchedule <= 5) Then
                    .AddItem "2 every section"
                    .ItemData(.NewIndex) = 5
                End If
                If (varDoorData.TrussSchedule = 0) Then
                    .AddItem "1 top only"
                    .ItemData(.NewIndex) = 6
                End If
                If (.ListCount > 0) Then
                    .ListIndex = 0
                    If ProdType = "Rolltite" Then
                        If DrType = "SR24" Then
                            If DrWidth >= 147 And DrWidth <= 242 Then
                                For i = 0 To .ListCount - 1
                                    If .ItemData(i) = 3 Then
                                        .ListIndex = i
                                    End If
                                Next
                            End If
                            If DrWidth > 242 And DrWidth <= 290 Then
                                For i = 0 To .ListCount - 1
                                    If .ItemData(i) = 5 Then
                                        .ListIndex = i
                                    End If
                                Next
                            End If
                        ElseIf DrType = "SR16" Or DrType = "SR1626" Then
                            If DrWidth >= 216 Then
                                For i = 0 To .ListCount - 1
                                    If .ItemData(i) = 3 Then
                                        .ListIndex = i
                                    End If
                                Next
                            End If
                        ElseIf DrType = "SR20" Then
                            If DrWidth >= 193 And DrWidth <= 242 Then
                                For i = 0 To .ListCount - 1
                                    If .ItemData(i) = 3 Then
                                        .ListIndex = i
                                    End If
                                Next
                            End If
                            If DrWidth > 242 And DrWidth <= 290 Then
                                For i = 0 To .ListCount - 1
                                    If .ItemData(i) = 5 Then
                                        .ListIndex = i
                                    End If
                                Next
                            End If
                        End If
                    End If
                End If
            Else
                MsgBox "Unable to find a truss schedule for this door." & vbCrLf & _
                    "Please try using a different door model, switch to Linear face hardware or consult with manufacturer for truss schedule.", vbApplicationModal + vbInformation, "Truss"
            End If
            
            End With
        Else 'NewLinear HW
            If cboWindLoad.ListIndex > 0 Then V = cboWindLoad.ItemData(cboWindLoad.ListIndex)
            Call SetDefaultZTrussDetails(V)
            If cboTrussStyle.ListCount > 0 Then cboTrussStyle.ListIndex = 0
        End If
    Else 'special truss
        Dim lobjFrmX As frmSecSpecialTruss
        Set lobjFrmX = New frmSecSpecialTruss
        lobjFrmX.Show vbModal
        Set lobjFrmX = Nothing
    End If
    
    Exit Sub
Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    Call DisplayError(llngErrNo, IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource, lstrErrDescr)

End Sub

Private Sub cmdBack_Click()

    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".cmdBack_Click()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    llngErrNo = 0
    lstrErrDescr = ""
    Dim lstrMsg As String, lintBut As Integer, lintRep As Integer
    
    
''''''''''Change in .NET''''''''''''''''
    Call RollBack("Sections", False, varPassDoor.strPN)
''''''''''END Change in .NET
    
    lstrMsg = "Are you sure you want to return to the previous screen?" & vbCrLf & _
        "Your selections made on this screen might be lost."
    lintBut = 4 '(yes,no)
    
    If frmGrid.txtSectionsNotes.Text <> "" Then
        lstrMsg = "By returning to the previous screen you might loose the selections made on this screen." & vbCrLf & _
            "Do you wish to clear the Sections Notes?"
        lintBut = 3 '(yes,no,cancel)
    End If
    
    lintRep = MsgBox(lstrMsg, vbApplicationModal + vbInformation + lintBut, "Return to the Previous Screen.")
    
    If lintRep = vbCancel Then Exit Sub
    
    If lintRep = vbYes Or (lintBut = 3 And lintRep = vbNo) Then
        
        If lintRep = vbYes Then frmGrid.txtSectionsNotes.Text = ""
        
        Screen.MousePointer = vbHourglass
        If Not isEmptyArraySpItems Then
            Dim larrTemp() As SpecialItem
            Dim i As Integer, UpperBound As Integer, lintCntr As Integer
            For i = 0 To UBound(varSpItem())
                If varSpItem(i).intType = 1 Then 'lite
                    ReDim Preserve larrTemp(lintCntr)
                    larrTemp(lintCntr).intType = varSpItem(i).intType
                    larrTemp(lintCntr).sngLength = varSpItem(i).sngLength
                    larrTemp(lintCntr).sngPrice = varSpItem(i).sngPrice
                    larrTemp(lintCntr).sngQty = varSpItem(i).sngQty
                    larrTemp(lintCntr).sngShippingWgt = varSpItem(i).sngShippingWgt
                    larrTemp(lintCntr).sngSpringingWgt = varSpItem(i).sngSpringingWgt
                    larrTemp(lintCntr).strDescription = varSpItem(i).strDescription
                    larrTemp(lintCntr).strNotes = varSpItem(i).strNotes
                    larrTemp(lintCntr).strPN = varSpItem(i).strPN
                    larrTemp(lintCntr).strUM = varSpItem(i).strUM
                    lintCntr = lintCntr + 1
                End If
            Next
            ReDim varSpItem(0)
            Erase varSpItem
            If lintCntr > 0 Then
                ReDim varSpItem(UBound(larrTemp))
                For i = 0 To UBound(larrTemp())
                    varSpItem(i).intType = larrTemp(i).intType
                    varSpItem(i).sngLength = larrTemp(i).sngLength
                    varSpItem(i).sngPrice = larrTemp(i).sngPrice
                    varSpItem(i).sngPrice = larrTemp(i).sngPrice

Private Sub AddZTrussSelectionToList(ByVal ZTrussType As Integer, ByVal ZTrussSelection As Integer)
    
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".AddZTrussSelectionToList()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    
    Dim TI As Integer, TempIndex As Integer, lstrCboText As String
    
    If ZTrussSelection = NoTruss Then
        TI = 0
    Else
        TI = ((ZTrussType - 1) * 100) + ZTrussSelection
    End If
    
    Select Case ZTrussSelection
        Case NoTruss
            lstrCboText = "No Truss"
        Case TopBottom
            lstrCboText = "Top and bottom"
        Case TopBottom2nd
            lstrCboText = "Top, btm & every 2nd sect."
        Case EverySect
            lstrCboText = "Every section"
    End Select
    
    If (Len(ZTrussSelection) > 0) Then
        If (ZTrussSelection <> NoTruss) Then
            If (ZTrussType = T25x18) Then
                lstrCboText = "T2.5 x 18 " & lstrCboText
            ElseIf (ZTrussType = T3625x18) Then
                lstrCboText = "T3.625 x 18 " & lstrCboText
            ElseIf (ZTrussType = T40x16) Then
                lstrCboText = "T4.0 x 16 " & lstrCboText
            End If
            cboTrussStyle.AddItem lstrCboText
            cboTrussStyle.ItemData(cboTrussStyle.NewIndex) = TI
        Else
            For TempIndex = 0 To 1
                Select Case TempIndex
                    Case 0
                        lstrCboText = "No truss"
                        TI = 0
                    Case Else
                        lstrCboText = "T2.5 x 18 Top sect. only"
                        TI = -1
                End Select
                cboTrussStyle.AddItem lstrCboText
                cboTrussStyle.ItemData(cboTrussStyle.NewIndex) = TI
            Next TempIndex
        End If
    End If

    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Sub

Public Sub SetDefaultZTrussDetails(ByVal PsfSel As Integer)
    
    On Error GoTo Error_Handler
    Const cstrErrorSource As String = cstrApplicationName & "." & cstrModuleName & ".SetDefaultZTrussDetails()"
    Dim llngErrNo As Long
    Dim lstrErrDescr As String
    Dim lstrErrSource As String
    llngErrNo = 0
    lstrErrDescr = ""
    lstrErrSource = ""
    'PURPOSE OF THE SUB:
    'This SUB sets available truss selection that can be applied to a door.
    'The calculation is based on the disciplines exclusively for Z-trusses that was developed
    'by RW in the summer of 1998.
    'Ray had asked to include the new Z-trusses in Aug 1998.
    'Disciplines supplied by RW. There are Categories for the selection:
    '   1) Basic
    '   2) 15 Psf design
    '   3) 20 Psf design
    '---------------------------------------------------------------------------
    'PARAMETERS NEEDED:
    '   1) PsfSel       : Wind load presure (Pounds per square foot)
    '---------------------------------------------------------------------------
    'RETURNED DATA:
    '   1) The settings on the combo box passed as parameter
    
    'NOTES: Maximum choice is 1 truss per section. This results in 4 choices for each door type
    '       per each Z-truss type.
    '       There are 3 Z-truss types:
    '           1) 0 = T-2.5   x 18 => T25x18
    '           2) 1 = T-3.625 x 18 => T3625x18
    '           3) 2 = T-4.0   x 16 => T40x16
    '       Choices per each Z-truss type
    '           a) 0 = No truss                                     => No truss
    '           b) 1 = Top and Bottom only.                         => Top and bottom
    '           c) 2 = Top, Bottom and every 2nd section from top.  => Top, bottom and every 2nd sect
    '           d) 3 = Top, Bottom and every section.               => Every section
    '
    '   The combo box will have proper dispcriptions displayed and the DATA property will store
    '   an integer with the following format:
    '       a) The value is always in the form of hundreds
    '       b) The hundred part is the z-truss type
    '       c) The part that is less than 10 is the choices for the assigned z-truss type
    '       d) The 0 value means that no truss needed
    
    'Local variables needed for the calculation.
    Dim S As String, TI As Integer
    cboTrussStyle.Enabled = True
    'Clear the contents of the combo box
    cboTrussStyle.Clear
    If cboTrussStyle.ListCount > 0 Then cboTrussStyle.ListIndex = 0
    
    If Not ((ProdType = "Thermatite") Or (ProdType = "Thermalume") Or (ProdType = "Replacement")) Then Exit Sub
    
    'Start the selection
    If (PsfSel = 0) Then            'Basic (75 M/hr failure rating rate)
        Call SetZTrussBasic
    ElseIf (PsfSel = 1) Then       '15 P/sf (90 M/hr failure rating rate)
        Call SetZTruss15Psf
    ElseIf (PsfSel = 2) Then       '20 P/sf (105 M/hr failure rating rate)
        Call SetZTruss20Psf
    End If
    If (cboTrussStyle.ListCount = 0) Then
        MsgBox "Unable to find a Z-Truss schedule for this door." & vbCrLf & _
                "Please consult with manufacturer for truss schedule.", vbApplicationModal + vbInformation, "Truss"
        cboTrussStyle.Enabled = False
        chkTrussUpgrade.Value = 0
        chkTrussUpgrade.Enabled = False
    Else
        cboTrussStyle.Enabled = True
        chkTrussUpgrade.Enabled = True
    End If
    Exit Sub

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Sub

'            lstrcutInstructions(i) = lstrcutInstructions(i) & frmGrid.Grid1.TextMatrix(frmGrid.Grid1.Row, 4)
            lintQty(i) = lintQty(i) + frmGrid.Grid1.TextMatrix(frmGrid.Grid1.Row, 1)
            i = i + 1
        End If
        If frmGrid.Grid1.Row < frmGrid.Grid1.Rows - 1 Then
            frmGrid.Grid1.Row = frmGrid.Grid1.Row + 1
        Else
            Exit Do
        End If
    Loop
       
    If lintQty(0) > 0 Then
        If lstrPartNum = "328-648" Then
            Call RollBack("Sections", True, "328-648")  'T150
            If MeasU = 1 Then
                Call FillSummaryGrid("328-648", 2, "Sections", GetSubCategory("328-648"), 0, "Cut (2) to " & GetImpString(DrWidth - 2))
            Else
                ClearConObj
                aCon.Inch = DrWidth - 2
                aCon.mmcalc
                Call FillSummaryGrid("328-648", 2, "Sections", GetSubCategory("328-648"), 0, "Cut (2) to " & Round(aCon.mm, 0) & " mm")
            End If
        ElseIf lstrPartNum = "328-653" Then             'T175
            Call RollBack("Sections", True, "328-653")
            If MeasU = 1 Then
                Call FillSummaryGrid("328-653", 2, "Sections", GetSubCategory("328-653"), 0, "Cut (2) to " & GetImpString(DrWidth - 2))
            Else
                ClearConObj
                aCon.Inch = DrWidth - 2
                aCon.mmcalc
                Call FillSummaryGrid("328-653", 2, "Sections", GetSubCategory("328-653"), 0, "Cut (2) to " & Round(aCon.mm, 0) & " mm")
            End If
        ElseIf lstrPartNum = "328-217" Then
            Dim LenRet As Single
            Dim MetLenRet As Single
            
            LenRet = DrWidth
            ClearConObj
            aCon.Inch = LenRet
            aCon.mmcalc
            MetLenRet = aCon.mm
            
            Call RollBack("Sections", True, "328-217")
            If MeasU = 1 Then
                 Call FillSummaryGrid("328-217", LenRet * 2, "Sections", GetSubCategory("328-217"), LenRet * 2, "Make (2) " & GetImpString(LenRet))
            Else
                Call FillSummaryGrid("328-217", MetLenRet * 2, "Sections", GetSubCategory("328-217"), LenRet * 2, "Make (2) " & Round(MetLenRet, 0) & " mm")
            End If
        End If
    End If
    
    Exit Function

Error_Handler:
    llngErrNo = IIf(llngErrNo = 0, Err.Number, llngErrNo)
    lstrErrDescr = IIf(Len(lstrErrDescr) = 0, Err.Description, lstrErrDescr & vbCrLf & Err.Description)
    lstrErrSource = IIf(Len(Trim(Err.Source)) = 0, "", Err.Source & vbCrLf) & cstrErrorSource
    Call Err.Raise(llngErrNo, lstrErrSource, lstrErrDescr)

End Function
